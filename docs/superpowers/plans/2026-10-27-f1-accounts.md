# F1: Accounts Implementation Plan

> **For agentic workers:** implement this plan task-by-task with the `executing-plans` skill. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Kameron and Catherine each sign in with their own name and password, and each person's data lives in their own D1 database. Kameron's existing database and history are untouched. Catherine sees a "plan on the way" Today until her plan exists.

**Architecture:**
- **Databases.** `rocky-db` stays Kameron's (`DB`). A new `rocky-db-catherine` (`DB_CATHERINE`) uses the same migrations. A new `rocky-core` (`CORE`) holds the household, users and login lockouts.
- **Request scope.** Root middleware resolves the user from the session cookie and runs the request inside an `AsyncLocalStorage` scope. `userDb()` returns that user's Drizzle client, so the 23 `getDb(env.DB)` call sites change and no query does.
- **Outside a request.** Cron and the steps ingest pick the database from the user row.

**Tech Stack:** React Router 8.4 middleware, Cloudflare Workers (`nodejs_als` flag), D1 + Drizzle 0.45, Web Crypto PBKDF2, Vitest 4 + `@cloudflare/vitest-plugin`, tsx for the CLI.

**Spec:** `docs/superpowers/specs/2026-10-27-foundation-design.md`, section 1 and rollout step 1. **Branch:** `v2/foundation`.

- Env and bindings: `import { env } from "cloudflare:workers"`. In tests: `import { env } from "cloudflare:test"`.
- Workers tests share storage within a file and start clean per file. Migrations are applied by `tests/workers/apply-migrations.ts`.
- No em dashes anywhere (code, comments, copy, commits). No `Co-Authored-By` lines in commits.
- `npm run dev` is the way to look at the app. Do not use `npm run build` to check UI.
- One full `npm test` at the end of the plan (Task 11), not after every task. Each task runs only its own test files.

// No Worker imports: scripts/user.ts runs this in Node to hash passwords before writing them to D1.

// The Workers runtime caps PBKDF2 at 100,000 iterations.
const ITERATIONS = 100_000;
const enc = new TextEncoder();

const toB64 = (bytes: Uint8Array) => btoa(String.fromCharCode(...bytes));
const fromB64 = (s: string) => Uint8Array.from(atob(s), (c) => c.charCodeAt(0));

async function derive(password: string, salt: Uint8Array, iterations: number): Promise<Uint8Array> {
  const key = await crypto.subtle.importKey("raw", enc.encode(password), "PBKDF2", false, ["deriveBits"]);
  return new Uint8Array(await crypto.subtle.deriveBits({ name: "PBKDF2", hash: "SHA-256", salt, iterations }, key, 256));
}

export async function hashPassword(password: string): Promise<string> {
  const salt = crypto.getRandomValues(new Uint8Array(16));
  return `pbkdf2$${ITERATIONS}$${toB64(salt)}$${toB64(await derive(password, salt, ITERATIONS))}`;
}

export async function verifyPassword(password: string, stored: string | null): Promise<boolean> {
  const parts = stored?.split("$");
  if (!parts || parts.length !== 4 || parts[0] !== "pbkdf2") {
    // Same work as a real check, so a missing user takes as long as a wrong password.
    await derive(password, new Uint8Array(16), ITERATIONS);
    return false;
  }
  const got = await derive(password, fromB64(parts[2]), Number(parts[1]));
  const want = fromB64(parts[3]);
  if (got.length !== want.length) return false;
  let diff = 0;
  for (let i = 0; i < got.length; i++) diff |= got[i] ^ want[i];
  return diff === 0;
}
```

import { env } from "cloudflare:workers";
import { and, eq, gte } from "drizzle-orm";
import { drizzle } from "drizzle-orm/d1";
import { getDb, type Db } from "~/db/client";
import * as core from "~/db/core-schema";

export type PersonBinding = "DB" | "DB_CATHERINE";

export interface User {
  id: number;
  name: string;
  dbBinding: PersonBinding;
  planKey: string | null;
  ingestToken: string | null;
}

export const coreDb = () => drizzle(env.CORE, { schema: core });

const columns = {
  id: core.users.id,
  name: core.users.name,
  dbBinding: core.users.dbBinding,
  planKey: core.users.planKey,
  ingestToken: core.users.ingestToken,
};

type Row = { id: number; name: string; dbBinding: string; planKey: string | null; ingestToken: string | null };
const asUser = (row: Row | undefined): User | null => (row ? { ...row, dbBinding: row.dbBinding as PersonBinding } : null);

export async function userById(id: number): Promise<User | null> {
  return asUser(await coreDb().select(columns).from(core.users).where(eq(core.users.id, id)).get());
}

export async function userByBinding(binding: PersonBinding): Promise<User | null> {
  return asUser(await coreDb().select(columns).from(core.users).where(eq(core.users.dbBinding, binding)).get());
}

export async function userByIngestToken(token: string): Promise<User | null> {
  if (!token) return null;
  return asUser(await coreDb().select(columns).from(core.users).where(eq(core.users.ingestToken, token)).get());
}

export async function allUsers(): Promise<User[]> {
  const rows = await coreDb().select(columns).from(core.users).orderBy(core.users.id).all();
  return rows.map((r) => asUser(r)!);
}

export async function credentialsFor(name: string): Promise<{ user: User; passwordHash: string | null } | null> {
  const row = await coreDb()
    .select({ ...columns, passwordHash: core.users.passwordHash })
    .from(core.users)
    .where(eq(core.users.nameKey, name.trim().toLowerCase()))
    .get();
  if (!row) return null;
  const { passwordHash, ...user } = row;
  return { user: asUser(user)!, passwordHash };
}

/** The Drizzle client for one person's own database. */
export function personDb(user: User): Db {
  return getDb(env[user.dbBinding]);
}

const WINDOW_MS = 15 * 60 * 1000;
const MAX_FAILURES = 5;

export async function isLockedOut(ip: string, now = new Date()): Promise<boolean> {
  const since = new Date(now.getTime() - WINDOW_MS).toISOString();
  const rows = await coreDb()
    .select({ id: core.loginAttempts.id })
    .from(core.loginAttempts)
    .where(and(eq(core.loginAttempts.ip, ip), gte(core.loginAttempts.attemptedAt, since)))
    .all();
  return rows.length >= MAX_FAILURES;
}

export async function recordFailure(ip: string, now = new Date()): Promise<void> {
  await coreDb().insert(core.loginAttempts).values({ ip, attemptedAt: now.toISOString() });
}
```

const PUBLIC_PATHS = [/^\/login$/, /^\/api\/ingest\//];

export type Auth = { user: User; setCookie?: string } | null;

/** The signed-in user for a request. Public paths give null; anything else without a valid session redirects to /login. */
export async function authenticate(request: Request): Promise<Auth> {
  const pathname = new URL(request.url).pathname.replace(/\.data$/, "");
  if (PUBLIC_PATHS.some((re) => re.test(pathname))) return null;
  const session = await getSession(request.headers.get("Cookie"));
  const userId = session.get("userId");
  if (userId) {

    if (user) return { user };

    if (user) {
      session.unset("signedIn");
      session.set("userId", user.id);
      return { user, setCookie: await commitSession(session) };
    }
  }
  throw redirect("/login");
}

export function clientIp(request: Request): string {

- [ ] **Step 1: Write the script.** Create `scripts/user.ts`:

```ts
// npm run user -- set-password <name> [--remote]
// npm run user -- token <name> [--rotate] [--remote]
import { execFileSync } from "node:child_process";
import { createInterface } from "node:readline";
import { hashPassword } from "../app/server/password";

const USAGE = "Usage: npm run user -- set-password <name> [--remote] | token <name> [--rotate] [--remote]";
const [command, rawName = "", ...flags] = process.argv.slice(2);
const remote = flags.includes("--remote");
const nameKey = rawName.trim().toLowerCase();

// nameKey is the only user input that reaches SQL, and this keeps it to letters.
if (!["set-password", "token"].includes(command ?? "") || !/^[a-z]+$/.test(nameKey)) {
  console.error(USAGE);
  process.exit(1);
}

function d1<T>(sql: string): T[] {
  const out = execFileSync("npx", ["wrangler", "d1", "execute", "rocky-core", remote ? "--remote" : "--local", "--json", "--command", sql], {
    encoding: "utf8",
  });
  return (JSON.parse(out) as { results: T[] }[])[0]?.results ?? [];
}

function askHidden(question: string): Promise<string> {
  return new Promise((resolve) => {
    const rl = createInterface({ input: process.stdin, output: process.stdout, terminal: true });
    const rlWithOutput = rl as unknown as { _writeToOutput(s: string): void };
    const write = rlWithOutput._writeToOutput.bind(rl);
    // Echo the prompt but not the keystrokes.
    rlWithOutput._writeToOutput = (s) => {
      if (s.startsWith(question)) write(s);
    };
    rl.question(question, (answer) => {
      rl.close();
      process.stdout.write("\n");
      resolve(answer);
    });
  });
}

const where = `name_key = '${nameKey}'`;
const target = remote ? "production" : "local";
if (d1<{ id: number }>(`SELECT id FROM users WHERE ${where}`).length === 0) {
  console.error(`No user named ${rawName} in ${target}.`);
  process.exit(1);
}

if (command === "set-password") {
  const password = await askHidden("New password: ");
  const again = await askHidden("Again: ");
  if (password !== again) {
    console.error("The passwords don't match.");
    process.exit(1);
  }
  if (password.length < 8) {
    console.error("Use at least 8 characters.");
    process.exit(1);
  }
  d1(`UPDATE users SET password_hash = '${await hashPassword(password)}' WHERE ${where}`);
  console.log(`Password set for ${rawName} (${target}).`);
} else if (flags.includes("--rotate")) {
  const token = Array.from(crypto.getRandomValues(new Uint8Array(24)), (b) => b.toString(16).padStart(2, "0")).join("");
  d1(`UPDATE users SET ingest_token = '${token}' WHERE ${where}`);
  console.log(token);
} else {
  const [row] = d1<{ token: string | null }>(`SELECT ingest_token AS token FROM users WHERE ${where}`);
  console.log(row?.token ?? "No token yet. Run again with --rotate to make one.");
}
```
